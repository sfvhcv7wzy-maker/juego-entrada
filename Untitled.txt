<script>
(function () {
  // ================================================================
  //  1) PEGA AQUÍ LA URL DE TU APLICACIÓN WEB DE GOOGLE APPS SCRIPT
  //     (termina en /exec). Si la dejas vacía, la página funciona en
  //     modo de prueba y guarda solo en este navegador.
  // ================================================================
  var SCRIPT_URL = "https://script.google.com/macros/s/AKfycbyIuC2bpV9lTeM2LUHVcNPmo2n0mbJh2CckeM7FPYdhISnY_SeFEji-r1Hnjmc2k3gh-A/exec";

  // ================================================================
  //  2) PARTIDOS: edita equipos, fechas y sedes. Cada id debe ser único
  //     (letras y números, sin espacios). Fecha: AAAA-MM-DDTHH:MM:00
  // ================================================================
  var JUEGOS = [
    { id: "g1", deporte: "Basketball", local: "Golden State", visitante: "Lakers", fecha: "2026-10-24T19:30:00", sede: "Arena Central", tipo: "cancha" },
    { id: "g2", deporte: "Futbol", local: "América", visitante: "Chivas", fecha: "2026-11-01T17:00:00", sede: "Estadio Norte", tipo: "campo" },
    { id: "g3", deporte: "Basketball", local: "Lakers", visitante: "Golden State", fecha: "2026-11-13T20:00:00", sede: "Arena Central", tipo: "cancha" },
    { id: "g4", deporte: "Futbol", local: "Tigres", visitante: "Monterrey", fecha: "2026-11-21T19:00:00", sede: "Estadio Norte", tipo: "campo" }
  ];

  var FILAS = ["A", "B", "C", "D", "E", "F"];
  var POR_FILA = 10;
  var TOTAL = FILAS.length * POR_FILA;
  var LS_KEY = "registro-entradas-web-v1";

  var juegoId = JUEGOS[0].id;
  var asiento = null;
  var tomados = {};
  var cargando = !!SCRIPT_URL;
  var errorRed = false;
  var registrando = false;

  function $(s) { return document.querySelector(s); }
  function h(tag, attrs) {
    var e = document.createElement(tag);
    attrs = attrs || {};
    Object.keys(attrs).forEach(function (k) {
      var v = attrs[k];
      if (k === "class") e.className = v;
      else if (k.indexOf("on") === 0) e.addEventListener(k.slice(2), v);
      else if (v === true) e.setAttribute(k, "");
      else if (v !== false && v != null) e.setAttribute(k, v);
    });
    for (var i = 2; i < arguments.length; i++) { if (arguments[i] != null && arguments[i] !== false) e.append(arguments[i]); }
    return e;
  }
  function nombreJuego(j) { return j.deporte + ": " + j.local + " vs " + j.visitante; }
  function juegoActual() { return JUEGOS.filter(function (j) { return j.id === juegoId; })[0]; }
  function ocupado(jid, as) { return !!tomados[jid + "_" + as]; }

  // ---- Modo de prueba (sin Google Sheets) ----
  function leerLocal() { try { var t = localStorage.getItem(LS_KEY); return t ? JSON.parse(t) : []; } catch (e) { return []; } }
  function guardarLocal(lista) { try { localStorage.setItem(LS_KEY, JSON.stringify(lista)); } catch (e) {} }
  function tomadosLocales() {
    var t = {};
    leerLocal().forEach(function (r) { t[r.juegoId + "_" + r.asiento] = true; });
    return t;
  }

  function canchaSVG(tipo) {
    if (tipo === "campo") {
      return '<svg viewBox="0 0 300 120" role="img" aria-label="Campo de futbol"><rect x="1" y="1" width="298" height="118" rx="6" fill="var(--campo)"/>' +
        '<g fill="none" stroke="#fff" stroke-width="2" opacity=".9"><rect x="8" y="8" width="284" height="104"/><line x1="150" y1="8" x2="150" y2="112"/>' +
        '<circle cx="150" cy="60" r="19"/><rect x="8" y="30" width="46" height="60"/><rect x="246" y="30" width="46" height="60"/>' +
        '<rect x="8" y="46" width="18" height="28"/><rect x="274" y="46" width="18" height="28"/></g></svg>';
    }
    return '<svg viewBox="0 0 300 120" role="img" aria-label="Cancha de basketball"><rect x="1" y="1" width="298" height="118" rx="6" fill="var(--court)"/>' +
      '<g fill="none" stroke="#fff" stroke-width="2" opacity=".9"><rect x="8" y="8" width="284" height="104"/><line x1="150" y1="8" x2="150" y2="112"/>' +
      '<circle cx="150" cy="60" r="18"/><rect x="8" y="36" width="54" height="48"/><rect x="238" y="36" width="54" height="48"/>' +
      '<circle cx="62" cy="60" r="13"/><circle cx="238" cy="60" r="13"/></g></svg>';
  }

  function aviso(texto, error) { var a = $("#aviso"); a.textContent = texto; a.className = "aviso " + (error ? "err" : "ok"); }
  function limpiarAviso() { var a = $("#aviso"); a.textContent = ""; a.className = "aviso"; }

  function renderJuegos() {
    $("#juegos").replaceChildren.apply($("#juegos"), JUEGOS.map(function (j) {
      var f = new Date(j.fecha);
      var activo = j.id === juegoId;
      var libres = TOTAL - Object.keys(tomados).filter(function (k) { return k.indexOf(j.id + "_") === 0; }).length;
      var mes = f.toLocaleDateString("es", { month: "short" }).replace(".", "");
      var dia = f.toLocaleDateString("es", { weekday: "long" });
      var hora = f.toLocaleTimeString("es", { hour: "2-digit", minute: "2-digit", hour12: false });
      return h("button", {
        class: "ticket" + (activo ? " activo" : ""), type: "button", "aria-pressed": String(activo),
        onclick: function () { juegoId = j.id; asiento = null; limpiarAviso(); render(); $("#asientos").scrollIntoView(); }
      },
        h("span", { class: "stub" }, h("span", { class: "dia" }, String(f.getDate())), h("span", { class: "mes" }, mes)),
        h("span", { class: "info" },
          h("span", { class: "deporte" }, j.deporte),
          h("span", { class: "equipos" }, j.local + " vs " + j.visitante),
          h("span", { class: "detalle" }, dia + " a las " + hora),
          h("span", { class: "detalle" }, j.sede)),
        h("span", { class: "libres" }, h("strong", {}, String(libres)), "libres"));
    }));
  }

  function renderMapa() {
    var j = juegoActual();
    var f = new Date(j.fecha);
    $("#sub-juego").textContent = nombreJuego(j) + ", " + f.toLocaleDateString("es", { day: "numeric", month: "long", year: "numeric" });
    var cancha = h("div", { class: "cancha" });
    cancha.innerHTML = canchaSVG(j.tipo);
    var leyenda = h("div", { class: "leyenda" },
      h("span", {}, h("i", { class: "muestra" }), "Libre"),
      h("span", {}, h("i", { class: "muestra ocu" }), "Ocupado"),
      h("span", {}, h("i", { class: "muestra mio" }), "Tu asiento"));
    var grid = h("div", { class: "tribuna", role: "group", "aria-label": "Mapa de asientos" });
    FILAS.forEach(function (fila) {
      grid.append(h("span", { class: "fila" }, fila));
      for (var n = 1; n <= POR_FILA; n++) {
        if (n === 6) grid.append(h("span", {}));
        (function (id) {
          var tomado = ocupado(j.id, id), sel = id === asiento;
          grid.append(h("button", {
            type: "button", class: "asiento" + (tomado ? " ocupado" : "") + (sel ? " sel" : ""),
            disabled: tomado, "aria-pressed": String(sel),
            "aria-label": "Asiento " + id + ", " + (tomado ? "ocupado" : "libre"),
            onclick: function () { asiento = id; limpiarAviso(); renderMapa(); renderResumen(); }
          }, id.slice(1)));
        })(fila + n);
      }
    });
    $("#mapa").replaceChildren(cancha, leyenda, grid);
  }

  function renderResumen() {
    $("#resumen").textContent = asiento ? "Asiento " + asiento + " en " + nombreJuego(juegoActual()) : "Toca un asiento libre para elegirlo.";
    var b = $("#registrar");
    b.disabled = registrando || cargando;
    b.textContent = cargando ? "Conectando…" : registrando ? "Registrando…" : "Registrar entrada";
    var t = "";
    if (!SCRIPT_URL) t = "Modo de prueba: la página aún no está conectada a Google Sheets, los registros se guardan solo en este navegador.";
    else if (errorRed) t = "No se pudieron cargar los asientos ocupados. Revisa tu conexión.";
    $("#nota").textContent = t;
  }

  function render() { renderJuegos(); renderMapa(); renderResumen(); }

  async function refrescar() {
    if (!SCRIPT_URL) { tomados = tomadosLocales(); cargando = false; render(); return; }
    try {
      var r = await fetch(SCRIPT_URL + "?t=" + Date.now());
      var d = await r.json();
      if (!d.ok) throw new Error("respuesta");
      tomados = {};
      d.ocupados.forEach(function (k) { tomados[k] = true; });
      errorRed = false;
      if (asiento && ocupado(juegoId, asiento)) {
        aviso("El asiento " + asiento + " acaba de ser ocupado. Elige otro.", true);
        asiento = null;
      }
    } catch (e) { errorRed = true; }
    cargando = false;
    render();
  }

  async function registrar() {
    if (registrando || cargando) return;
    var input = $("#nombre");
    var nombre = input.value.trim().replace(/\s+/g, " ");
    if (!nombre) { aviso("Escribe tu nombre para registrar la entrada.", true); input.focus(); return; }
    if (!asiento) { aviso("Elige un asiento libre en el mapa.", true); return; }
    var j = juegoActual(), as = asiento;
    registrando = true; renderResumen();
    try {
      var resultado = "ok";
      if (!SCRIPT_URL) {
        var lista = leerLocal();
        if (ocupado(j.id, as)) resultado = "tomado";
        else { lista.push({ juegoId: j.id, asiento: as, nombre: nombre, ts: Date.now() }); guardarLocal(lista); }
      } else {
        var resp = await fetch(SCRIPT_URL, {
          method: "POST",
          headers: { "Content-Type": "text/plain;charset=utf-8" },
          body: JSON.stringify({ juegoId: j.id, juego: nombreJuego(j), asiento: as, nombre: nombre })
        });
        var d = await resp.json();
        resultado = d.ok ? "ok" : (d.error === "tomado" ? "tomado" : "error");
      }
      if (resultado === "ok") {
        tomados[j.id + "_" + as] = true;
        asiento = null; input.value = "";
        aviso("Entrada registrada: " + nombre + ", asiento " + as + ", " + nombreJuego(j) + ".", false);
      } else if (resultado === "tomado") {
        asiento = null;
        aviso("El asiento " + as + " ya está ocupado. Elige otro.", true);
        refrescar();
      } else {
        aviso("No se pudo guardar la entrada. Intenta de nuevo.", true);
      }
    } catch (e) {
      aviso("No se pudo conectar. Revisa tu internet e intenta de nuevo.", true);
    }
    registrando = false;
    render();
  }

  $("#registrar").addEventListener("click", registrar);
  $("#nombre").addEventListener("keydown", function (e) { if (e.key === "Enter") registrar(); });

  render();
  refrescar();
  if (SCRIPT_URL) {
    setInterval(function () { if (document.visibilityState === "visible" && !registrando) refrescar(); }, 30000);
  }
})();
</script>
